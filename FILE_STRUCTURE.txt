🎯 QUICK COMMAND REFERENCE

Installation:
  python -m venv venv
  venv\Scripts\activate          # Windows
  source venv/bin/activate       # Linux/Mac
  pip install -r requirements.txt

Configuration:
  cp .env.example .env
  # Edit .env with your credentials

Run:
  python main.py

Test:
  python tests.py
  pytest tests.py -v             # With pytest

Docker:
  docker-compose build
  docker-compose up -d
  docker-compose logs -f

Deploy:
  # See DEPLOYMENT.md for systemd, cloud, VPS guides
